local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local StarterGui = game:GetService("StarterGui")

local player = Players.LocalPlayer
local folder = workspace:WaitForChild("_LocalGoldNuggetDrops")

local running = false

--==================================================
-- NOTIFICATION
--==================================================

local function notify(text)
	pcall(function()
		StarterGui:SetCore("SendNotification", {
			Title = "Gold Nugget",
			Text = text,
			Duration = 3
		})
	end)

	print("[Gold Nugget] " .. text)
end

--==================================================
-- FIND DROPS
--==================================================

local function getDrops()
	local drops = {}

	for _, drop in ipairs(folder:GetChildren()) do
		if drop:FindFirstChild("PickupAnchor", true) then
			table.insert(drops, drop)
		end
	end

	return drops
end

--==================================================
-- COLLECT ONE — RETRY UNTIL SUCCESSFUL
--==================================================

local function collectDrop(drop, index, total)

	local character =
		player.Character
		or player.CharacterAdded:Wait()

	local attempts = 0

	while running and drop.Parent do

		attempts += 1

		local anchor = drop:FindFirstChild("PickupAnchor", true)

		if not anchor then
			task.wait(0.25)
			continue
		end

		-- Teleport to the nugget
		character:PivotTo(
			anchor.CFrame + Vector3.new(0, 3, 0)
		)

		task.wait(0.25)

		-- Find the ProximityPrompt
		local prompt =
			anchor:FindFirstChildWhichIsA("ProximityPrompt", true)

		if not prompt then
			prompt = drop:FindFirstChildWhichIsA(
				"ProximityPrompt",
				true
			)
		end

		if prompt then

			if attempts == 1 or attempts % 5 == 0 then
				notify(
					"Trying nugget "
					.. index
					.. "/"
					.. total
					.. " — attempt "
					.. attempts
				)
			end

			print(
				"[Gold Nugget] Attempting:",
				index,
				"Attempt:",
				attempts,
				"Prompt:",
				prompt.Name
			)

			-- Activate the game's normal ProximityPrompt
			pcall(function()
				fireproximityprompt(prompt)
			end)

		else

			if attempts == 1 or attempts % 5 == 0 then
				warn(
					"[Gold Nugget] No ProximityPrompt found for nugget "
					.. index
					.. " — retrying..."
				)
			end

		end

		-- Give the game time to process the pickup
		task.wait(0.5)

	end

	return not drop.Parent
end

--==================================================
-- COLLECTION LOOP
--==================================================

local function collectAll()

	if running then
		return
	end

	running = true

	local drops = getDrops()

	notify(
		"Found "
		.. #drops
		.. " gold nuggets"
	)

	for i, drop in ipairs(drops) do

		if not running then
			break
		end

		notify(
			"Going to nugget "
			.. i
			.. "/"
			.. #drops
		)

		local success = collectDrop(
			drop,
			i,
			#drops
		)

		if success then
			notify(
				"Collected "
				.. i
				.. "/"
				.. #drops
			)
		end

	end

	running = false

	notify("Collection run finished")
end

--==================================================
-- ' TO START
--==================================================

UserInputService.InputBegan:Connect(function(input, processed)

	if processed then
		return
	end

	if input.KeyCode == Enum.KeyCode.Quote then
		task.spawn(collectAll)
	end

end)

--==================================================
-- RESPAWN DETECTION
--==================================================

folder.ChildAdded:Connect(function(drop)

	task.wait(0.1)

	if drop:FindFirstChild("PickupAnchor", true) then
		notify("A gold nugget respawned!")
	end

end)

--==================================================
-- READY
--==================================================

print("[Gold Nugget] Collector loaded")
notify("Collector ready — press '")
